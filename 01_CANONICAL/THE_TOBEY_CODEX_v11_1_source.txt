// Constitutional: the core computation engine
pub fn reiterate(
    start: Position,
    aspects: &[AspectOp]
) -> ReiterationResult {
    let mut current = start;
    let mut total_atp = 0.0;
    let mut history = Vec::new();
    let mut iteration = 0;

    while iteration < MAX_ITERATIONS {
        history.push(current.sign_index);

        // Apply aspect sequence
        for aspect in aspects {
            let (new_position, atp_cost) = current.apply_aspect(*aspect);
            current = new_position;
            total_atp += atp_cost;
        }

        iteration += 1;

        // Check convergence
        if let Some(conv_type) = detect_convergence(&history) {
            return ReiterationResult {
                converged: true,
                iterations: iteration,
                final_position: current,
                net_atp: total_atp,
                convergence_type: Some(conv_type),
                cipher_address: current.sign_index,
                precision_level: PrecisionLevel::Pythagorean,
            };
        }
    }

    // Divergent
    ReiterationResult {
        converged: false,
        iterations: MAX_ITERATIONS,
        final_position: current,
        net_atp: total_atp,
        convergence_type: None,
        cipher_address: current.sign_index,
        precision_level: PrecisionLevel::Geometric,
    }
}

fn detect_convergence(history: &[usize]) -> Option<ConvergenceType> {
    if history.len() < 2 {
        return None;
    }

    let last = *history.last().unwrap();
    let prev = *history.get(history.len() - 2).unwrap();

    // Fixed point (Conjunction pattern)
    if last == prev {
        return Some(ConvergenceType::FixedPoint);
    }

    // 2-cycle (Opposition pattern)
    if history.len() >= 4 {
        let pattern = [
            history[history.len() - 4],
            history[history.len() - 3],
            history[history.len() - 2],
            history[history.len() - 1],
        ];
        if pattern[0] == pattern[2] && pattern[1] == pattern[3] && pattern[0] != pattern[1] {
            return Some(ConvergenceType::TwoCycle);
        }
    }

    // 4-cycle (Square pattern)
    if history.len() >= 8 {
        let cycle_start = history[history.len() - 4];
        let all_match = history[history.len() - 8..history.len() - 4]
            .iter()
            .zip(history[history.len() - 4..].iter())
            .all(|(a, b)| a == b);

        if all_match && cycle_start != last {
            return Some(ConvergenceType::FourCycle);
        }
    }

    // Stable triangle (Trine pattern)
    if history.len() >= 6 {
        let cycle_start = history[history.len() - 3];
        let all_match = history[history.len() - 6..history.len() - 3]
            .iter()
            .zip(history[history.len() - 3..].iter())
            .all(|(a, b)| a == b);

        if all_match && cycle_start != last {
            return Some(ConvergenceType::StableTriangle);
        }
    }

    None
}

// Constitutional: the core computation engine
pub fn reiterate(
    start: Position,
    aspects: &[AspectOp]
) -> ReiterationResult {
    let mut current = start;
    let mut total_atp = 0.0;
    let mut history = Vec::new();
    let mut iteration = 0;

    while iteration < MAX_ITERATIONS {
        history.push(current.sign_index);

        // Apply aspect sequence
        for aspect in aspects {
            let (new_position, atp_cost) = current.apply_aspect(*aspect);
            current = new_position;
            total_atp += atp_cost;
        }

        iteration += 1;

        // Check convergence
        if let Some(conv_type) = detect_convergence(&history) {
            return ReiterationResult {
                converged: true,
                iterations: iteration,
                final_position: current,
                net_atp: total_atp,
                convergence_type: Some(conv_type),
                cipher_address: current.sign_index,
                precision_level: PrecisionLevel::Pythagorean,
            };
        }
    }

    // Divergent
    ReiterationResult {
        converged: false,
        iterations: MAX_ITERATIONS,
        final_position: current,
        net_atp: total_atp,
        convergence_type: None,
        cipher_address: current.sign_index,
        precision_level: PrecisionLevel::Geometric,
    }
}

fn detect_convergence(history: &[usize]) -> Option<ConvergenceType> {
    if history.len() < 2 {
        return None;
    }

    let last = *history.last().unwrap();
    let prev = *history.get(history.len() - 2).unwrap();

    // Fixed point (Conjunction pattern)
    if last == prev {
        return Some(ConvergenceType::FixedPoint);
    }

    // 2-cycle (Opposition pattern)
    if history.len() >= 4 {
        let pattern = [
            history[history.len() - 4],
            history[history.len() - 3],
            history[history.len() - 2],
            history[history.len() - 1],
        ];
        if pattern[0] == pattern[2] && pattern[1] == pattern[3] && pattern[0] != pattern[1] {
            return Some(ConvergenceType::TwoCycle);

pub fn compute_timestamp() -> String {
    let now = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .unwrap()
